如何在自己的页面中使用这些组件

1. 首先，在页面中引入组件的JavaScript文件：

<script src="components/header.js" defer></script>
<script src="components/footer.js" defer></script>

2. 在HTML中添加容器元素：

<div id="header-container"></div>
<!-- 页面内容 -->
<div id="footer-container"></div>

3. 在页面加载完成后初始化组件：

<script>
    document.addEventListener('DOMContentLoaded', () => {
        // 初始化header组件
        const headerContainer = document.getElementById('header-container');
        if (headerContainer && window.HeaderComponent) {
            const header = new HeaderComponent();
            headerContainer.appendChild(header.create());
            
            // 可选：自定义用户信息
            header.updateUserInfo('李业主');
            // header.updateUserAvatar('path/to/custom/avatar.jpg');
        }
        
        // 初始化footer组件
        const footerContainer = document.getElementById('footer-container');
        if (footerContainer && window.FooterComponent) {
            const footer = new FooterComponent();
            footerContainer.appendChild(footer.create());
            
            // 可选：自定义公司信息
            // footer.updateCompanyInfo('您的物业公司', '2023');
            // footer.updateContactInfo('010-12345678', 'info@yourcompany.com', '您的地址');
        }
    });
</script>
